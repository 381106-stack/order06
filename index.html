教學重點與核心演算法解析

本範例為高三「專題實作」與「程式設計演算法」綜合示範。核心理念為將複雜的問題拆解為**資料結構 (Data Structure)** 與 **邏輯控制 (Control Logic)**。

1. 資料結構設計

使用 JavaScript 物件陣列 (Array of Objects) 存放菜單與訂單資料，實現資料與畫面的解耦。

2. 輸入驗證 (Validation)

防呆機制 (Guard Clauses)：座號限制 1~50、餐點限制 1~5，輸入不合法時拒絕寫入且不讓系統崩潰。

3. 統計與累加 (Aggregation)

使用高階陣列函式 (`reduce` / `forEach`)，進行全班總金額與各品項小計的快速計算。

core_logic.js (原生 JavaScript 示範) 

// ============================================================
// 班級午餐訂餐系統 - 核心邏輯範例 (高三資訊科實作)
// ============================================================

// 1. 菜單資料結構 (內建 5 項菜單)
const MENU = [
    { id: 1, name: "排骨便當", price: 80 },
    { id: 2, name: "雞腿便當", price: 85 },
    { id: 3, name: "素食便當", price: 70 },
    { id: 4, name: "滷肉飯",   price: 55 },
    { id: 5, name: "貢丸湯",   price: 20 }
];

// 2. 儲存全班訂單資料的陣列
let orderList = [];

/**
 * 驗證座號與餐點編號是否合法
 * @param {number} seat - 座號 (應為 1~50)
 * @param {number} itemNo - 餐點編號 (應為 1~5)
 * @returns {object} { isValid: boolean, errorMsg: string }
 */
function validateInput(seat, itemNo) {
    if (isNaN(seat) || seat < 1 || seat > 50) {
        return { isValid: false, errorMsg: "座號不合法！必須為 1 到 50 之間的數字。" };
    }
    if (isNaN(itemNo) || itemNo < 1 || itemNo > MENU.length) {
        return { isValid: false, errorMsg: `餐點編號不合法！必須為 1 到 ${MENU.length} 之間的數字。` };
    }
    return { isValid: true, errorMsg: "" };
}

/**
 * 新增一筆訂單
 */
function addOrder(seat, itemNo) {
    const validation = validateInput(seat, itemNo);
    if (!validation.isValid) {
        return validation;
    }

    const menuItem = MENU.find(item => item.id === itemNo);
    const newOrder = {
        orderId: Date.now(),
        seatNumber: seat,
        itemId: menuItem.id,
        itemName: menuItem.name,
        price: menuItem.price
    };

    orderList.push(newOrder);
    return { isValid: true, order: newOrder };
}

/**
 * 計算全班結算統計報表
 */
function calculateSummary() {
    // 初始化各品項統計
    const summaryMap = {};
    MENU.forEach(item => {
        summaryMap[item.id] = { name: item.name, price: item.price, count: 0, subtotal: 0 };
    });

    let totalAmount = 0;

    // 累加數量與金額
    orderList.forEach(order => {
        if (summaryMap[order.itemId]) {
            summaryMap[order.itemId].count += 1;
            summaryMap[order.itemId].subtotal += order.price;
            totalAmount += order.price;
        }
    });

    return {
        itemSummaries: summaryMap,
        totalOrders: orderList.length,
        totalAmount: totalAmount
    };
}

/**
 * 題目需求：控制台模擬輸入 (座號 0 終止迴圈)
 */
function runSimulatedLoop() {
    while (true) {
        let seatInput = prompt("請輸入學生座號 (1-50)，輸入 0 結束登記：");
        if (seatInput === null) break; // 使用者按下取消
        let seat = parseInt(seatInput, 10);

        if (seat === 0) {
            alert("已輸入座號 0，結束訂餐登記流程！");
            break;
        }

        let itemInput = prompt("請輸入餐點編號 (1:排骨, 2:雞腿, 3:素食, 4:滷肉飯, 5:貢丸湯)：");
        if (itemInput === null) break;
        let itemNo = parseInt(itemInput, 10);

        const result = addOrder(seat, itemNo);
        if (!result.isValid) {
            alert("❌ " + result.errorMsg);
        } else {
            alert(`✅ 成功登記：${seat} 號訂購 ${result.order.itemName}`);
        }
    }
}
